SQL：品类月度 GMV 趋势 展开 
SELECT 
    p.category,
    DATE_FORMAT(s.date, '%Y-%m') AS month,
    SUM(s.qty) AS total_qty,
    SUM(s.gmv) AS total_gmv,
    COUNT(DISTINCT s.sku) AS sku_count
FROM sales_daily s
JOIN products p ON s.sku = p.sku
WHERE s.date BETWEEN '2026-07-01' AND '2026-09-30'
GROUP BY p.category, DATE_FORMAT(s.date, '%Y-%m')
ORDER BY p.category, month;
SQL：Q3 TOP 10 SKU 排名 展开 
SELECT 
    p.sku,
    p.sku_name,
    p.category,
    SUM(s.qty) AS q3_qty,
    SUM(s.gmv) AS q3_gmv
FROM sales_daily s
JOIN products p ON s.sku = p.sku
WHERE s.date BETWEEN '2026-07-01' AND '2026-09-30'
GROUP BY p.sku
ORDER BY q3_gmv DESC
LIMIT 10;

SQL：供应商综合绩效评估（三层 CTE + 窗口函数） 展开 
WITH supplier_base AS (
    SELECT 
        supplier_id,
        COUNT(DISTINCT sku) AS sku_count,
        COUNT(*) AS total_orders,
        ROUND(SUM(CASE WHEN actual_arrival <= expected_arrival THEN 1 ELSE 0 END) 
              * 100.0 / COUNT(*), 1) AS on_time_rate,
        ROUND((SUM(qty) - SUM(qc_defect_qty)) * 100.0 / SUM(qty), 1) AS quality_rate
    FROM purchase_orders
    GROUP BY supplier_id
),
supplier_gmv AS (
    SELECT p.supplier_id, SUM(s.gmv) AS q3_gmv
    FROM products p
    JOIN sales_daily s ON p.sku = s.sku
    WHERE s.date BETWEEN '2026-07-01' AND '2026-09-30'
    GROUP BY p.supplier_id
),
combined AS (
    SELECT 
        b.supplier_id, b.sku_count, b.total_orders,
        b.on_time_rate, b.quality_rate, g.q3_gmv,
        ROUND(g.q3_gmv * 100.0 / SUM(g.q3_gmv) OVER(), 1) AS gmv_share_pct,
        ROUND(b.on_time_rate * 0.40 + b.quality_rate * 0.40 
              + (g.q3_gmv * 100.0 / SUM(g.q3_gmv) OVER()) * 0.20, 2) AS composite_score
    FROM supplier_base b
    JOIN supplier_gmv g ON b.supplier_id = g.supplier_id
)
SELECT supplier_id, sku_count, total_orders, on_time_rate, quality_rate,
       q3_gmv, gmv_share_pct, composite_score,
       RANK() OVER(ORDER BY composite_score DESC) AS supplier_rank
FROM combined
ORDER BY supplier_rank;

SQL：多时间窗口销量聚合 展开 
WITH sales_90d AS (
    SELECT p.sku, COALESCE(SUM(s.qty), 0) AS qty_90d
    FROM products p
    LEFT JOIN sales_daily s ON p.sku = s.sku 
        AND s.date BETWEEN '2026-07-02' AND '2026-09-30'
    GROUP BY p.sku
),
sales_60d AS (
    SELECT p.sku, COALESCE(SUM(s.qty), 0) AS qty_60d
    FROM products p
    LEFT JOIN sales_daily s ON p.sku = s.sku 
        AND s.date BETWEEN '2026-08-02' AND '2026-09-30'
    GROUP BY p.sku
),
sales_30d AS (
    SELECT p.sku, COALESCE(SUM(s.qty), 0) AS qty_30d
    FROM products p
    LEFT JOIN sales_daily s ON p.sku = s.sku 
        AND s.date BETWEEN '2026-09-01' AND '2026-09-30'
    GROUP BY p.sku
)
SELECT ... FROM products p
JOIN sales_90d s90 ON p.sku = s90.sku
JOIN sales_60d s60 ON p.sku = s60.sku
JOIN sales_30d s30 ON p.sku = s30.sku;
SQL：库存状态与趋势判断 展开 
CASE 
    WHEN avg_daily_30d = 0 AND current_stock > 0 THEN '滞销(有库无销)'
    WHEN avg_daily_30d = 0 AND current_stock = 0 THEN '断货(无库无销)'
    WHEN current_stock / avg_daily_30d < 7 THEN '库存不足'
    WHEN current_stock / avg_daily_30d BETWEEN 7 AND 30 THEN '库存健康'
    ELSE '库存积压'
END AS stock_status,

CASE 
    WHEN avg_daily_30d > avg_daily_60d AND avg_daily_60d > avg_daily_90d THEN '上升趋势'
    WHEN avg_daily_30d < avg_daily_60d AND avg_daily_60d < avg_daily_90d THEN '下降趋势'
    ELSE '趋势平稳'
END AS trend_signal
